def oracle(ref='control_0:v1'):
    return {'signal_ref':ref,'oracle_status':'sufficient_unique','expected_state':'neutral',
        'unit':'dimensionless ratio','expected_numeric':{'decimal':'0.25'},
        'measurement_period':{'start':None,'end':'2024-12-31'},'operands':[],
        'numeric_comparison_policy':{'raw_numeric_absolute_tolerance':'1e-12'}}


def native(index):
    return SignalResult(f'control_{index}',1,SignalValue.NEUTRAL,0.5,
        f'control_{index}=0.25; synthetic only',measurement=NumericMeasurement(0.25,'ratio','synthetic',period_end='2024-12-31'))


class SafetyControls(unittest.TestCase):
    def test_exact_unit_period_numeric_and_ambiguity(self):
        expected, result = oracle(),native(0).to_dict()
        self.assertTrue(m.compare(expected,result)['passed'])
        for changed in (
            {**result,'measurement':{**result['measurement'],'value':0.3}},
            {**result,'measurement':{**result['measurement'],'unit':'fraction'}},
            {**result,'measurement':{**result['measurement'],'period':{'end':'2023-12-31'}}},
            {**result,'state':{'value':'positive'}},
        ):
            self.assertFalse(m.compare(expected,changed)['passed'])
        self.assertIsNone(m.compare({'signal_ref':'x:v1','oracle_status':'period_policy_ambiguous'},result)['passed'])

    def test_operand_whole_literal_not_substring(self):
        expected,result=oracle(),native(0).to_dict()
        expected['operands']=[{'document_id':'source','literal':'{"value":10}','fact':{'tag':'Cash','value':10}}]
        result['evidence']['supporting']=[{'document_id':'source','exact_span':'prefix {"value":10}'}]
        self.assertFalse(m.compare(expected,result)['passed'])
        result['evidence']['supporting'][0]['exact_span']='{"value":10}'
        self.assertTrue(m.compare(expected,result)['passed'])

    def test_registry_requires_existing_dirs_and_detects_constructor_write(self):
        with TemporaryDirectory() as temp:
            root=Path(temp); registry=root/'registry'; candidates=root/'candidates'
            group={'root':str(registry),'version':12,'refs':['x']*18,'registry_hash':'hash'}
            with patch.object(m,'SignalRegistry') as ctor:
                with self.assertRaises(ValueError): m.readonly_snapshot(group,candidates)
                ctor.assert_not_called()
            registry.mkdir(); candidates.mkdir(); m.write_new(registry/'registry.json',{'version':12})
            snap=SimpleNamespace(registry_hash='hash',specs=tuple(range(18)))
            with patch.object(m,'SignalRegistry',return_value=SimpleNamespace(snapshot_from_refs=lambda *_:snap)):
                self.assertIs(m.readonly_snapshot(group,candidates),snap)
            def mutate(*a,**kw):
                m.write_new(candidates/'unexpected.json',{})
                return SimpleNamespace(snapshot_from_refs=lambda *_:snap)
            with patch.object(m,'SignalRegistry',side_effect=mutate):
                with self.assertRaisesRegex(ValueError,'changed tree'):m.readonly_snapshot(group,candidates)

    def test_fixed_destination_and_invalid_plan_write_nothing(self):
        with TemporaryDirectory() as temp:
            root=Path(temp); fixed=root/'stage'; wrong=root/'protected'/'new'
            with patch.object(m,'OUTPUT',fixed),patch.object(m,'authorities') as authority:
                with self.assertRaises(ValueError):m.prepare(wrong)
                authority.assert_not_called();self.assertFalse(wrong.exists())
                with self.assertRaises(FileNotFoundError):m.execute(fixed,'bad')
                self.assertFalse(fixed.exists())
