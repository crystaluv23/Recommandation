def node(text, name):
    return next(n for n in ast.parse(text).body if isinstance(n, (ast.FunctionDef, ast.ClassDef)) and n.name == name)

def unchanged(path, name):
    old = subprocess.check_output(['git', 'show', 'HEAD:'+path], cwd=ROOT, text=True)
    new = (ROOT/path).read_text()
    assert ast.dump(node(old, name), include_attributes=False) == ast.dump(node(new, name), include_attributes=False)
    return {'path': path, 'definition': name, 'ast_equal_to_HEAD': True}
